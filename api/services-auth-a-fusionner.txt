import {
  ConflictException,
  Injectable,
} from '@nestjs/common';

import {
  DataSource,
  QueryFailedError,
} from 'typeorm';

import {
  ApiCodeResponse,
  ApiException,
} from '../../../common/api/index.js';

import {
  AccountEntity,
  AccountResponseDto,
  AccountStatus,
} from '../../../core/account/data/index.js';

import {
  AuthIdentifierEntity,
  AuthIdentifierType,
  PasswordCredentialEntity,
  RegisterRequestDto,
} from '../data/index.js';

import {
  AuthIdentifierService,
} from './auth-identifier.service.js';

import {
  PasswordHasherService,
} from './password-hasher.service.js';

@Injectable()
export class RegistrationService {
  constructor(
    private readonly dataSource:
      DataSource,

    private readonly passwordHasherService:
      PasswordHasherService,
      
    private readonly authIdentifierService:
      AuthIdentifierService,
  ) {}

  async register(
    input: RegisterRequestDto,
  ): Promise<AccountResponseDto> {
    const normalizedEmail =
      this.authIdentifierService
        .normalizeEmail(
          input.email,
        );

    const passwordHash =
      await this.passwordHasherService
        .hashPassword(
          input.password,
        );

    try {
      const account =
        await this.dataSource.transaction(
          async (
            transactionalManager,
          ): Promise<AccountEntity> => {
            const identifierRepository =
              transactionalManager.getRepository(
                AuthIdentifierEntity,
              );

            const existingIdentifier =
              await identifierRepository
                .findOne({
                  where: {
                    type:
                      AuthIdentifierType.Email,

                    normalizedValue:
                      normalizedEmail,
                  },
                });

            if (existingIdentifier) {
              throw new ConflictException();
            }

            const accountRepository =
              transactionalManager.getRepository(
                AccountEntity,
              );

            const credentialRepository =
              transactionalManager.getRepository(
                PasswordCredentialEntity,
              );

            const accountEntity =
              accountRepository.create({
                status:
                  AccountStatus.Pending,
              });

            const savedAccount =
              await accountRepository.save(
                accountEntity,
              );

            const identifier =
              identifierRepository.create({
                accountId:
                  savedAccount.id,

                type:
                  AuthIdentifierType.Email,

                value:
                  input.email.trim(),

                normalizedValue:
                  normalizedEmail,

                verifiedAt: null,
              });

            await identifierRepository.save(
              identifier,
            );

            const credential =
              credentialRepository.create({
                accountId:
                  savedAccount.id,

                passwordHash,

                passwordChangedAt:
                  new Date(),
              });

            await credentialRepository.save(
              credential,
            );

            return savedAccount;
          },
        );

      return AccountResponseDto.fromEntity(
        account,
      );
    } catch (error: unknown) {
      if (
        error instanceof
        ConflictException
      ) {
        throw this.createIdentifierConflict();
      }

      if (
        this.isUniqueViolation(error)
      ) {
        throw this.createIdentifierConflict();
      }

      throw error;
    }
  }

  private createIdentifierConflict():
    ApiException<null> {
    return new ApiException({
      statusCode: 409,

      code:
        ApiCodeResponse
          .AuthRegistrationIdentifierUnavailable,

      data: null,

      logMessage:
        'Registration identifier is unavailable',
    });
  }

  private isUniqueViolation(
    error: unknown,
  ): boolean {
    if (
      !(
        error instanceof
        QueryFailedError
      )
    ) {
      return false;
    }

    const driverError =
      error.driverError as {
        code?: string;
      };

    return (
      driverError.code === '23505'
    );
  }
}

import {
  HttpStatus,
  Injectable,
} from '@nestjs/common';

import {
  InjectRepository,
} from '@nestjs/typeorm';

import {
  Repository,
} from 'typeorm';

import {
  ApiCodeResponse,
  ApiException,
} from '../../../common/api/index.js';

import {
  AccountEntity,
  AccountResponseDto,
  AccountStatus,
} from '../../../core/account/data/index.js';

import {
  PasswordCredentialService,
} from './password-credential.service.js';

import {
  LoginRequestDto,
  LoginResponseDto,
} from '../data/index.js';
import {
    type CreateSessionOptions,
    AuthSessionService,
} from './auth-session.service.js';
import {
  AuthIdentifierService,
} from './auth-identifier.service.js';
import {
  RefreshTokenService,
} from './refresh-token.service.js';

import {
  TokenCryptoService,
} from './token-crypto.service.js';
@Injectable()
export class LoginService {
  constructor(
    @InjectRepository(AccountEntity)
    private readonly accountRepository: Repository<AccountEntity>,
    private readonly passwordCredentialService: PasswordCredentialService,
    private readonly authAuthSessionService: AuthSessionService,
    private readonly authIdentifierService: AuthIdentifierService,
    private readonly refreshTokenService: RefreshTokenService,
    private readonly tokenCryptoService: TokenCryptoService,
  ) {}

  async login(
  input: LoginRequestDto,
  sessionOptions:
    CreateSessionOptions = {},
): Promise<LoginResponseDto> {
  const identifier =
    await this.authIdentifierService
      .findEmailIdentifier(
        input.email,
      );

  if (!identifier) {
    throw this
      .createInvalidCredentialsException();
  }

  const account =
    await this.accountRepository.findOne({
      where: {
        id: identifier.accountId,
      },
    });

  if (
    !account ||
    account.status ===
      AccountStatus.Disabled
  ) {
    throw this
      .createInvalidCredentialsException();
  }

  const passwordIsValid =
    await this.passwordCredentialService
      .verifyPassword(
        account.id,
        input.password,
      );

  if (!passwordIsValid) {
    throw this
      .createInvalidCredentialsException();
  }

  const session =
    await this.authAuthSessionService
      .createSession(
        account.id,
        sessionOptions,
      );

  const tokenPair =
    await this.tokenCryptoService
      .generateTokenPair(
        account.id,
        session.id,
      );

  await this.refreshTokenService
    .createRefreshToken(
      session.id,
      tokenPair.refreshToken,
      tokenPair.refreshTokenExpiresAt,
    );

  const response =
    new LoginResponseDto();

  response.account =
    AccountResponseDto.fromEntity(
      account,
    );

  response.sessionId =
    session.id;

  response.sessionExpiresAt =
    session.expiresAt.toISOString();

  response.accessToken =
    tokenPair.accessToken;

  response.accessTokenExpiresInSeconds =
    tokenPair.accessTokenExpiresInSeconds;

  response.refreshToken =
    tokenPair.refreshToken;

  response.refreshTokenExpiresAt =
    tokenPair.refreshTokenExpiresAt
      .toISOString();

  return response;
}
  private createInvalidCredentialsException():
    ApiException<null> {
    return new ApiException<null>({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthLoginInvalidCredentials,

      data: null,

      logMessage:
        'Authentication failed because the credentials are invalid',
    });
  }
}

import {
  Injectable,
} from '@nestjs/common';

import {
  DataSource,
} from 'typeorm';

import {
  AuthRefreshTokenEntity,
  AuthSessionEntity,
  LogoutRequestDto,
  LogoutResponseDto,
} from '../data/index.js';

import {
  TokenCryptoService,
} from './token-crypto.service.js';

@Injectable()
export class LogoutService {
  constructor(
    private readonly dataSource:
      DataSource,

    private readonly tokenCryptoService:
      TokenCryptoService,
  ) {}

  async logout(
    input: LogoutRequestDto,
  ): Promise<LogoutResponseDto> {
    const tokenHash =
      this.tokenCryptoService
        .hashRefreshToken(
          input.refreshToken,
        );

    await this.dataSource.transaction(
      async (
        transactionalManager,
      ): Promise<void> => {
        const refreshTokenRepository =
          transactionalManager.getRepository(
            AuthRefreshTokenEntity,
          );

        const sessionRepository =
          transactionalManager.getRepository(
            AuthSessionEntity,
          );

        const refreshToken =
          await refreshTokenRepository
            .createQueryBuilder(
              'refreshToken',
            )
            .addSelect(
              'refreshToken.tokenHash',
            )
            .setLock(
              'pessimistic_write',
            )
            .where(
              'refreshToken.tokenHash = :tokenHash',
              {
                tokenHash,
              },
            )
            .getOne();

        /*
         * La déconnexion reste volontairement
         * idempotente. Un token inconnu ne révèle
         * aucune information au client.
         */
        if (!refreshToken) {
        return;
        }

        const session =
          await sessionRepository
            .createQueryBuilder('session')
            .setLock('pessimistic_write')
            .where(
              'session.id = :sessionId',
              {
                sessionId:
                  refreshToken.sessionId,
              },
            )
            .getOne();

        const now = new Date();

        await refreshTokenRepository
          .createQueryBuilder()
          .update(AuthRefreshTokenEntity)
          .set({
            revokedAt: now,
          })
          .where(
            'session_id = :sessionId',
            {
            sessionId:
                refreshToken.sessionId,
            },         )
          .andWhere(
         'revoked_at IS NULL',
        )
          .execute();

     if (
          session &&
      !session.revokedAt
        ){
          session.revokedAt = now;

          await sessionRepository.save(
            session,
        );
        }
      },
    );

  const response =
      new LogoutResponseDto();

    response.loggedOut = true;

    return response; }
}

import {
  HttpStatus,
  Injectable,
} from '@nestjs/common';

import {
  DataSource,
 
} from 'typeorm';
import type{
    Repository,
} from 'typeorm'
import {
  ApiCodeResponse,
  ApiException,
} from '../../../common/api/index.js';

import {
  AuthRefreshTokenEntity,
  AuthSessionEntity,
  RefreshRequestDto,
  RefreshResponseDto,
} from '../data/index.js';

import {
  TokenCryptoService,
} from './token-crypto.service.js';

type RotationResult =
  | {
      status: 'success';
      response: RefreshResponseDto;
    }
  | {
      status: 'invalid';
    }
  | {
      status: 'reuse-detected';
    };

@Injectable()
export class TokenRotationService {
  constructor(
    private readonly dataSource:
      DataSource,

    private readonly tokenCryptoService:
      TokenCryptoService,
  ) {}

  async rotate(
    input: RefreshRequestDto,
  ): Promise<RefreshResponseDto> {
    const currentTokenHash =
      this.tokenCryptoService
        .hashRefreshToken(
          input.refreshToken,
        );

    const result =
      await this.dataSource.transaction(
        async (
          transactionalManager,
        ): Promise<RotationResult> => {
          const refreshTokenRepository =
            transactionalManager.getRepository(
              AuthRefreshTokenEntity,
            );

          const sessionRepository =
            transactionalManager.getRepository(
              AuthSessionEntity,
            );

          const currentToken =
            await refreshTokenRepository
              .createQueryBuilder(
                'refreshToken',
              )
              .addSelect(
                'refreshToken.tokenHash',
              )
              .setLock(
                'pessimistic_write',
              )
              .where(
                'refreshToken.tokenHash = :tokenHash',
                {
                  tokenHash:
                    currentTokenHash,
                },
              )
              .getOne();

          /*
           * Token inconnu :
           * aucune session identifiable,
           * donc aucune révocation possible.
           */
          if (!currentToken) {
            return {
              status: 'invalid',
            };
          }

          /*
           * Un token déjà utilisé ou remplacé
           * est une réutilisation suspecte.
           */
          if (
            currentToken.usedAt ||
            currentToken.replacedByTokenId
          ) {
            await this.revokeSessionFamily(
              currentToken.sessionId,
              refreshTokenRepository,
              sessionRepository,
            );

            return {
              status:
                'reuse-detected',
            };
          }

          /*
           * Un token explicitement révoqué
           * ou expiré est simplement invalide.
           */
          if (
            currentToken.revokedAt ||
            currentToken.expiresAt
              .getTime() <= Date.now()
          ) {
            return {
              status: 'invalid',
            };
          }

          const session =
            await sessionRepository
              .createQueryBuilder(
                'session',
              )
              .setLock(
                'pessimistic_write',
              )
              .where(
                'session.id = :sessionId',
                {
                  sessionId:
                    currentToken.sessionId,
                },
              )
              .getOne();

          if (
            !session ||
            session.revokedAt ||
            session.expiresAt
              .getTime() <= Date.now()
          ) {
            return {
              status: 'invalid',
            };
          }

          const tokenPair =
            await this.tokenCryptoService
              .generateTokenPair(
                session.accountId,
                session.id,
              );

          const replacementToken =
            refreshTokenRepository.create({
              sessionId:
                session.id,

              tokenHash:
                tokenPair
                  .refreshTokenHash,

              expiresAt:
                tokenPair
                  .refreshTokenExpiresAt,

              usedAt: null,
              revokedAt: null,

              replacedByTokenId:
                null,
            });

          const savedReplacementToken =
            await refreshTokenRepository
              .save(
                replacementToken,
              );

          currentToken.usedAt =
            new Date();

          currentToken.replacedByTokenId =
            savedReplacementToken.id;

          await refreshTokenRepository.save(
            currentToken,
          );

          session.lastSeenAt =
            new Date();

          await sessionRepository.save(
            session,
          );

          const response =
            new RefreshResponseDto();

          response.accessToken =
            tokenPair.accessToken;

          response
            .accessTokenExpiresInSeconds =
            tokenPair
              .accessTokenExpiresInSeconds;

          response.refreshToken =
            tokenPair.refreshToken;

          response.refreshTokenExpiresAt =
            tokenPair
              .refreshTokenExpiresAt
              .toISOString();

          return {
            status: 'success',
            response,
          };
        },
      );

    if (
      result.status ===
      'reuse-detected'
    ) {
      throw this
        .createReuseDetectedException();
    }

    if (
      result.status ===
      'invalid'
    ) {
      throw this
        .createInvalidTokenException();
    }

    return result.response;
  }

  private async revokeSessionFamily(
  sessionId: string,
  refreshTokenRepository:
    Repository<AuthRefreshTokenEntity>,
  sessionRepository:
    Repository<AuthSessionEntity>,
): Promise<void> {
  const now = new Date();

  await refreshTokenRepository
    .createQueryBuilder()
    .update(
      AuthRefreshTokenEntity,
    )
    .set({
      revokedAt: now,
    })
    .where(
      'session_id = :sessionId',
      {
        sessionId,
      },
    )
    .andWhere(
      'revoked_at IS NULL',
    )
    .execute();

  const session =
    await sessionRepository
      .createQueryBuilder(
        'session',
      )
      .setLock(
        'pessimistic_write',
      )
      .where(
        'session.id = :sessionId',
        {
          sessionId,
        },
      )
      .getOne();

  if (
    session &&
    !session.revokedAt
  ) {
    session.revokedAt = now;

    await sessionRepository.save(
      session,
    );
  }
}

  private createInvalidTokenException():
    ApiException<null> {
    return new ApiException<null>({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthRefreshTokenInvalid,

      data: null,

      logMessage:
        'Refresh token validation failed',
    });
  }

  private createReuseDetectedException():
    ApiException<null> {
    return new ApiException<null>({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthRefreshTokenReuseDetected,

      data: null,

      logMessage:
        'Refresh token reuse was detected and the session was revoked',
    });
  }
}
