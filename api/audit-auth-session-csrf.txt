import {
  Injectable,
} from '@nestjs/common';

import {
  InjectRepository,
} from '@nestjs/typeorm';

import {
  Repository,
} from 'typeorm';

import {
  AuthSessionEntity,
} from '../data/index.js';

const SESSION_DURATION_DAYS = 30;

export type CreateSessionOptions = {
  ipAddress?: string | null;
  userAgent?: string | null;
  csrfTokenHash?: string;
};

@Injectable()
export class AuthSessionService {
  constructor(
    @InjectRepository(
      AuthSessionEntity,
    )
    private readonly sessionRepository:
      Repository<AuthSessionEntity>,
  ) {}

  async createSession(
    accountId: string,
    options: CreateSessionOptions = {},
  ): Promise<AuthSessionEntity> {
    const now = new Date();

    const expiresAt = new Date(
      now.getTime() +
        SESSION_DURATION_DAYS *
          24 *
          60 *
          60 *
          1000,
    );

    const session =
      this.sessionRepository.create({
        accountId,
        expiresAt,
        revokedAt: null,
        lastSeenAt: now,
        ipAddress:
          options.ipAddress ?? null,
        userAgent:
          options.userAgent ?? null,
          accessVersion:
            1,
      });

    return this.sessionRepository.save(
      session,
    );
  }

  async findActiveSession(
    sessionId: string,
  ): Promise<AuthSessionEntity | null> {
    const session =
      await this.sessionRepository.findOne({
        where: {
          id: sessionId,
        },
      });

    if (!session) {
      return null;
    }

    if (session.revokedAt) {
      return null;
    }

    if (
      session.expiresAt.getTime() <=
      Date.now()
    ) {
      return null;
    }

    return session;
  }

  async revokeSession(
    sessionId: string,
  ): Promise<boolean> {
    const session =
      await this.sessionRepository.findOne({
        where: {
          id: sessionId,
        },
      });

    if (!session) {
      return false;
    }

    if (session.revokedAt) {
      return true;
    }

    session.revokedAt = new Date();

    await this.sessionRepository.save(
      session,
    );

    return true;
  }
}

type RotationResult =
  | {
      status: 'success';
      response: RefreshResponseDto;
    }
  | {
      status: 'invalid';
    }
  | {
      status: 'reuse-detected';
    };

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(
      AccountEntity,
    )
    private readonly accountRepository:
      Repository<AccountEntity>,

    private readonly dataSource:
      DataSource,

    private readonly authIdentifierService:
      AuthIdentifierService,

    private readonly authSessionService:
      AuthSessionService,

    private readonly passwordCredentialService:
      PasswordCredentialService,

    private readonly passwordHasherService:
      PasswordHasherService,
    private readonly passwordPolicyService:
      PasswordPolicyService,
    private readonly refreshTokenService:
      RefreshTokenService,

    private readonly tokenCryptoService:
      TokenCryptoService,
  ) {}

  async register(
    input:
      RegisterRequestDto,
  ): Promise<AccountResponseDto> {
     const normalizedEmail =
                normalizeEmail(
                input.email,
            );   
            this.passwordPolicyService
            .assertValid(
            input.password,
            );
    const passwordHash =
      await this.passwordHasherService
        .hashPassword(
          input.password,
        );

    try {
      const account =
        await this.dataSource
          .transaction(
            async (
              transactionalManager,
            ): Promise<AccountEntity> => {
              const identifierRepository =
                transactionalManager
                  .getRepository(
                    AuthIdentifierEntity,
                  );

              const existingIdentifier =
                await identifierRepository
                  .findOne({
                    where: {
                      type:
                        AuthIdentifierType
                          .Email,

                      normalizedValue:
                        normalizedEmail,
                    },
                  });

              if (
                existingIdentifier
              ) {
                throw new ConflictException();
              }

              const accountRepository =
                transactionalManager
                  .getRepository(
                    AccountEntity,
                  );

              const credentialRepository =
                transactionalManager
                  .getRepository(
                    PasswordCredentialEntity,
                  );

              const accountEntity =
                accountRepository
                  .create({
                    status:
                      AccountStatus
                        .Pending,
                  });
                
              const savedAccount =
                await accountRepository
                  .save(
                    accountEntity,
                  );

              const identifier =
                identifierRepository
                  .create({
                    accountId:
                      savedAccount.id,

                    type:
                      AuthIdentifierType
                        .Email,

                    value:
                      input.email.trim(),

                    normalizedValue:
                      normalizedEmail,

                    verifiedAt:
                      null,
                      
                  });

              await identifierRepository
                .save(
                  identifier,
                );

              const credential =
                credentialRepository
                  .create({
                    accountId:
                      savedAccount.id,

                    passwordHash,

                    passwordChangedAt:
                      new Date(),
                  });

              await credentialRepository
                .save(
                  credential,
                );

              return savedAccount;
            },
          );

      return AccountResponseDto
        .fromEntity(
          account,
        );
    } catch (
      error: unknown
    ) {
      if (
        error instanceof
        ConflictException
      ) {
        throw this
          .createIdentifierConflictException();
      }

      if (
        this.isUniqueViolation(
          error,
        )
      ) {
        throw this
          .createIdentifierConflictException();
      }

      throw error;
    }
  }

  async login(
    input:
      LoginRequestDto,

    sessionOptions:
      CreateSessionOptions = {},
  ): Promise<LoginResponseDto> {
    const identifier =
      await this.authIdentifierService
        .findEmailIdentifier(
          input.email,
        );

    if (!identifier) {
      throw this
        .createInvalidCredentialsException();
    }

    const account =
      await this.accountRepository
        .findOne({
          where: {
            id:
              identifier.accountId,
          },
        });

    if (
      !account ||
      account.status ===
        AccountStatus.Disabled
    ) {
      throw this
        .createInvalidCredentialsException();
    }

    const passwordIsValid =
      await this
        .passwordCredentialService
        .verifyPassword(
          account.id,
          input.password,
        );

    if (!passwordIsValid) {
      throw this
        .createInvalidCredentialsException();
    }

    const session =
      await this.authSessionService
        .createSession(
          account.id,
          sessionOptions,
        );

    const tokenPair =
      await this.tokenCryptoService
        .generateTokenPair(
          account.id,
          session.id,
          session.accessVersion,
        );

    await this.refreshTokenService
      .createRefreshToken(
        session.id,
        tokenPair.refreshToken,
        tokenPair
          .refreshTokenExpiresAt,
      );

    const response =
      new LoginResponseDto();

    response.account =
      AccountResponseDto
        .fromEntity(
          account,
        );

    response.sessionId =
      session.id;

    response.sessionExpiresAt =
      session.expiresAt
        .toISOString();

    response.accessToken =
      tokenPair.accessToken;

    response
      .accessTokenExpiresInSeconds =
      tokenPair
        .accessTokenExpiresInSeconds;

    response.refreshToken =
      tokenPair.refreshToken;

    response.refreshTokenExpiresAt =
      tokenPair
        .refreshTokenExpiresAt
        .toISOString();

    return response;
  }

  async refresh(
    input:
      RefreshRequestDto,
  ): Promise<RefreshResponseDto> {
    const currentTokenHash =
      this.tokenCryptoService
        .hashRefreshToken(
          input.refreshToken,
        );

    const result =
      await this.dataSource
        .transaction(
          async (
            transactionalManager,
          ): Promise<RotationResult> => {
            const refreshTokenRepository =
              transactionalManager
                .getRepository(
                  AuthRefreshTokenEntity,
                );

            const sessionRepository =
              transactionalManager
                .getRepository(
                  AuthSessionEntity,
                );

            const currentToken =
              await refreshTokenRepository
                .createQueryBuilder(
                  'refreshToken',
                )
                .addSelect(
                  'refreshToken.tokenHash',
                )
                .setLock(
                  'pessimistic_write',
                )
                .where(
                  'refreshToken.tokenHash = :tokenHash',
                  {
                    tokenHash:
                      currentTokenHash,
                  },
                )
                .getOne();

            if (!currentToken) {
              return {
                status: 'invalid',
              };
            }

            if (
              currentToken.usedAt ||
              currentToken
                .replacedByTokenId
            ) {
              await this
                .revokeSessionFamily(
                  currentToken
                    .sessionId,

                  refreshTokenRepository,

                  sessionRepository,
                );

              return {
                status:
                  'reuse-detected',
              };
            }

            if (
              currentToken.revokedAt ||
              currentToken.expiresAt
                .getTime() <=
                Date.now()
            ) {
              return {
                status: 'invalid',
              };
            }

            const session =
              await sessionRepository
                .createQueryBuilder(
                  'session',
                )
                .setLock(
                  'pessimistic_write',
                )
                .where(
                  'session.id = :sessionId',
                  {
                    sessionId:
                      currentToken
                        .sessionId,
                  },
                )
                .getOne();

            if (
              !session ||
              session.revokedAt ||
              session.expiresAt
                .getTime() <=
                Date.now()
            ) {
              return {
                status:
                  'invalid',
              };
            }

            const tokenPair =
              await this
                .tokenCryptoService
                .generateTokenPair(
                  session.accountId,
                  session.id,
                  session.accessVersion,
                );

            const replacementToken =
              refreshTokenRepository
                .create({
                  sessionId:
                    session.id,

                  tokenHash:
                    tokenPair
                      .refreshTokenHash,

                  expiresAt:
                    tokenPair
                      .refreshTokenExpiresAt,

                  usedAt:
                    null,

                  revokedAt:
                    null,

                  replacedByTokenId:
                    null,
                });

            const savedReplacementToken =
              await refreshTokenRepository
                .save(
                  replacementToken,
                );

            currentToken.usedAt =
              new Date();

            currentToken
              .replacedByTokenId =
              savedReplacementToken.id;

            await refreshTokenRepository
              .save(
                currentToken,
              );

            session.lastSeenAt =
              new Date();

            await sessionRepository
              .save(
                session,
              );

            const response =
              new RefreshResponseDto();

            response.accessToken =
              tokenPair.accessToken;

            response
              .accessTokenExpiresInSeconds =
              tokenPair
                .accessTokenExpiresInSeconds;

            response.refreshToken =
              tokenPair.refreshToken;

            response
              .refreshTokenExpiresAt =
              tokenPair
                .refreshTokenExpiresAt
                .toISOString();

            return {
              status: 'success',
              response,
            };
          },
        );

    if (
      result.status ===
      'reuse-detected'
    ) {
      throw this
        .createReuseDetectedException();
    }

    if (
      result.status ===
      'invalid'
    ) {
      throw this
        .createInvalidRefreshTokenException();
    }

    return result.response;
  }

  async logout(
    input:
      LogoutRequestDto,
  ): Promise<LogoutResponseDto> {
    const tokenHash =
      this.tokenCryptoService
        .hashRefreshToken(
          input.refreshToken,
        );

    await this.dataSource
      .transaction(
        async (
          transactionalManager,
        ): Promise<void> => {
          const refreshTokenRepository =
            transactionalManager
              .getRepository(
                AuthRefreshTokenEntity,
              );

          const sessionRepository =
            transactionalManager
              .getRepository(
                AuthSessionEntity,
              );

          const refreshToken =
            await refreshTokenRepository
              .createQueryBuilder(
                'refreshToken',
              )
              .addSelect(
                'refreshToken.tokenHash',
              )
              .setLock(
                'pessimistic_write',
              )
              .where(
                'refreshToken.tokenHash = :tokenHash',
                {
                  tokenHash,
                },
              )
              .getOne();

          /*
           * La déconnexion est
           * volontairement idempotente.
           */
          if (!refreshToken) {
            return;
          }

          const session =
            await sessionRepository
              .createQueryBuilder(
                'session',
              )
              .setLock(
                'pessimistic_write',
              )
              .where(
                'session.id = :sessionId',
                {
                  sessionId:
                    refreshToken
                      .sessionId,
                },
              )
              .getOne();

          const now =
            new Date();

          await refreshTokenRepository
            .createQueryBuilder()
            .update(
              AuthRefreshTokenEntity,
            )
            .set({
              revokedAt:
                now,
            })
            .where(
              'session_id = :sessionId',
              {
                sessionId:
                  refreshToken
                    .sessionId,
              },
            )
            .andWhere(
              'revoked_at IS NULL',
            )
            .execute();

          if (
            session &&
            !session.revokedAt
          ) {
            session.revokedAt =
              now;

            await sessionRepository
              .save(
                session,
              );
          }
        },
      );

    const response =
      new LogoutResponseDto();

    response.loggedOut =
      true;

    return response;
  }

  private async revokeSessionFamily(
    sessionId: string,

    refreshTokenRepository:
      Repository<AuthRefreshTokenEntity>,

    sessionRepository:
      Repository<AuthSessionEntity>,
  ): Promise<void> {
    const now =
      new Date();

    await refreshTokenRepository
      .createQueryBuilder()
      .update(
        AuthRefreshTokenEntity,
      )
      .set({
        revokedAt:
          now,
      })
      .where(
        'session_id = :sessionId',
        {
          sessionId,
        },
      )
      .andWhere(
        'revoked_at IS NULL',
      )
      .execute();

    const session =
      await sessionRepository
        .createQueryBuilder(
          'session',
        )
        .setLock(
          'pessimistic_write',
        )
        .where(
          'session.id = :sessionId',
          {
            sessionId,
          },
        )
        .getOne();

    if (
      session &&
      !session.revokedAt
    ) {
      session.revokedAt =
        now;

      await sessionRepository
        .save(
          session,
        );
    }
  }

  private createIdentifierConflictException():
    ApiException<null> {
    return new ApiException({
      statusCode:
        HttpStatus.CONFLICT,

      code:
        ApiCodeResponse
          .AuthRegistrationIdentifierUnavailable,

      data:
        null,

      logMessage:
        'Registration identifier is unavailable',
    });
  }

  private createInvalidCredentialsException():
    ApiException<null> {
    return new ApiException({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthLoginInvalidCredentials,

      data:
        null,

      logMessage:
        'Authentication failed because the credentials are invalid',
    });
  }

  private createInvalidRefreshTokenException():
    ApiException<null> {
    return new ApiException({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthRefreshTokenInvalid,

      data:
        null,

      logMessage:
        'Refresh token validation failed',
    });
  }

  private createReuseDetectedException():
    ApiException<null> {
    return new ApiException({
      statusCode:
        HttpStatus.UNAUTHORIZED,

      code:
        ApiCodeResponse
          .AuthRefreshTokenReuseDetected,

      data:
        null,

      logMessage:
        'Refresh token reuse was detected and the session was revoked',
    });
  }

  private isUniqueViolation(
    error: unknown,
  ): boolean {
    if (
      !(
        error instanceof
        QueryFailedError
      )
    ) {
      return false;
    }

    const driverError =
      error.driverError as {
        code?: string;
      };

    return (
      driverError.code ===
      '23505'
    );
  }
}

===== TokenCryptoService =====
import {
  Injectable,
} from '@nestjs/common';

import {
  JwtService,
} from '@nestjs/jwt';

import {
  createHmac,
  randomBytes,
  randomUUID,
} from 'node:crypto';

import {
  EnvService,
} from '../../../common/config/index.js';

export type AccessTokenPayload = {
  sub: string;
  sid: string;
  jti: string;
  ver: number;
};

export type GeneratedTokenPair = {
  accessToken: string;
  accessTokenExpiresInSeconds: number;

  refreshToken: string;
  refreshTokenHash: string;
  refreshTokenExpiresAt: Date;
};

@Injectable()
export class TokenCryptoService {
  constructor(
    private readonly jwtService:
      JwtService,

    private readonly envService:
      EnvService,
  ) {}

  async generateTokenPair(
    accountId: string,
    sessionId: string,
    accessVersion: number,
  ): Promise<GeneratedTokenPair> {
    const accessTokenExpiresInSeconds =
      this.envService
        .authAccessTokenExpiresInSeconds;
    const tokenId = randomUUID();  
    const accessToken =
      await this.jwtService.signAsync(
        {
          sub: accountId,
          sid: sessionId,
          jti: randomUUID(),
          ver: accessVersion,
        } satisfies AccessTokenPayload,
        {
          secret:
            this.envService
              .authAccessTokenSecret,

          expiresIn:
            accessTokenExpiresInSeconds,
        },
      );

    const refreshToken =
      this.generateRefreshToken();

    const refreshTokenHash =
      this.hashRefreshToken(
        refreshToken,
      );

    const refreshTokenExpiresAt =
      this.calculateRefreshTokenExpiration();

    return {
      accessToken,
      accessTokenExpiresInSeconds,
      refreshToken,
      refreshTokenHash,
      refreshTokenExpiresAt,
    };
  }

  fingerprint(
  value: string,
): string {
  return createHmac(
    'sha256',
    this.envService
      .authRefreshTokenSecret,
  )
    .update(
      value,
    )
    .digest(
      'hex',
    );
}

hashRefreshToken(
  refreshToken: string,
): string {
  return this.fingerprint(
    refreshToken,
  );
}

  async verifyAccessToken(
    accessToken: string,
  ): Promise<AccessTokenPayload> {
    return this.jwtService
      .verifyAsync<AccessTokenPayload>(
        accessToken,
        {
          secret:
            this.envService
              .authAccessTokenSecret,
        },
      );
  }

  private generateRefreshToken():
    string {
    return randomBytes(48)
      .toString('base64url');
  }

  private calculateRefreshTokenExpiration():
    Date {
    const durationInMilliseconds =
      this.envService
        .authRefreshTokenExpiresInDays *
      24 *
      60 *
      60 *
      1000;

    return new Date(
      Date.now() +
        durationInMilliseconds,
    );
  }
}
