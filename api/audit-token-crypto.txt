===== TokenCryptoService =====
import {
  Injectable,
} from '@nestjs/common';

import {
  JwtService,
} from '@nestjs/jwt';

import {
  createHmac,
  randomBytes,
} from 'node:crypto';

import {
  EnvService,
} from '../../../common/config/index.js';

export type AccessTokenPayload = {
  sub: string;
  sid: string;
};

export type GeneratedTokenPair = {
  accessToken: string;
  accessTokenExpiresInSeconds: number;

  refreshToken: string;
  refreshTokenHash: string;
  refreshTokenExpiresAt: Date;
};

@Injectable()
export class TokenCryptoService {
  constructor(
    private readonly jwtService:
      JwtService,

    private readonly envService:
      EnvService,
  ) {}

  async generateTokenPair(
    accountId: string,
    sessionId: string,
  ): Promise<GeneratedTokenPair> {
    const accessTokenExpiresInSeconds =
      this.envService
        .authAccessTokenExpiresInSeconds;

    const accessToken =
      await this.jwtService.signAsync(
        {
          sub: accountId,
          sid: sessionId,
        } satisfies AccessTokenPayload,
        {
          secret:
            this.envService
              .authAccessTokenSecret,

          expiresIn:
            accessTokenExpiresInSeconds,
        },
      );

    const refreshToken =
      this.generateRefreshToken();

    const refreshTokenHash =
      this.hashRefreshToken(
        refreshToken,
      );

    const refreshTokenExpiresAt =
      this.calculateRefreshTokenExpiration();

    return {
      accessToken,
      accessTokenExpiresInSeconds,
      refreshToken,
      refreshTokenHash,
      refreshTokenExpiresAt,
    };
  }

  fingerprint(
  value: string,
): string {
  return createHmac(
    'sha256',
    this.envService
      .authRefreshTokenSecret,
  )
    .update(
      value,
    )
    .digest(
      'hex',
    );
}

hashRefreshToken(
  refreshToken: string,
): string {
  return this.fingerprint(
    refreshToken,
  );
}

  async verifyAccessToken(
    accessToken: string,
  ): Promise<AccessTokenPayload> {
    return this.jwtService
      .verifyAsync<AccessTokenPayload>(
        accessToken,
        {
          secret:
            this.envService
              .authAccessTokenSecret,
        },
      );
  }

  private generateRefreshToken():
    string {
    return randomBytes(48)
      .toString('base64url');
  }

  private calculateRefreshTokenExpiration():
    Date {
    const durationInMilliseconds =
      this.envService
        .authRefreshTokenExpiresInDays *
      24 *
      60 *
      60 *
      1000;

    return new Date(
      Date.now() +
        durationInMilliseconds,
    );
  }
}

===== AuthSessionEntity =====
import {
  Column,
  Entity,
  Index,
  JoinColumn,
  ManyToOne,
} from 'typeorm';

import {
  BasePersistenceEntity,
} from '../../../../common/database/index.js';

import {
  AccountEntity,
} from '../../../../core/account/data/index.js';

@Entity({
  name: 'auth_session',
})
@Index(
  'idx_auth_session_account_id',
  [
    'accountId',
  ],
)
@Index(
  'idx_auth_session_expires_at',
  [
    'expiresAt',
  ],
)
export class AuthSessionEntity
  extends BasePersistenceEntity
{
  @Column({
    name: 'account_id',
    type: 'varchar',
    length: 26,
  })
  accountId!: string;

  @ManyToOne(
    () => AccountEntity,
    {
      nullable: false,
      onDelete: 'CASCADE',
    },
  )
  @JoinColumn({
    name: 'account_id',
  })
  account!: AccountEntity;

  @Column({
    name: 'expires_at',
    type: 'timestamptz',
  })
  expiresAt!: Date;

  @Column({
    name: 'revoked_at',
    type: 'timestamptz',
    nullable: true,
  })
  revokedAt!: Date | null;

  @Column({
    name: 'last_seen_at',
    type: 'timestamptz',
    nullable: true,
  })
  lastSeenAt!: Date | null;

  @Column({
    name: 'ip_address',
    type: 'varchar',
    length: 45,
    nullable: true,
  })
  ipAddress!: string | null;
  @Column({
  name:
    'csrf_token_hash',

  type:
    'varchar',

  length:
    64,
})
csrfTokenHash!: string;
  @Column({
    name: 'user_agent',
    type: 'text',
    nullable: true,
  })
  userAgent!: string | null;
}
