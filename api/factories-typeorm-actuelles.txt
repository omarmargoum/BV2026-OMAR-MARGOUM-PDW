===== datasource-options.factory.ts =====
import type {
  DataSourceOptions,
} from 'typeorm';

import {
  AppMode,
  DatabaseType,
  EnvService,
} from '../config/index.js';

export const createDataSourceOptions = (
  envService: EnvService,
): DataSourceOptions => {
  const synchronize: boolean =
    envService.appMode === AppMode.Production
      ? false
      : envService.databaseSynchronize;

  return {
    type: DatabaseType.PostgreSql,
    host: envService.databaseHost,
    port: envService.databasePort,
    username: envService.databaseUsername,
    password: envService.databasePassword,
    database: envService.databaseName,
    synchronize,
  };
};

===== nest-typeorm-options.factory.ts =====
import type {
  TypeOrmModuleOptions,
} from '@nestjs/typeorm';

import {
  EnvService,
} from '../config/index.js';

import {
  createDataSourceOptions,
} from './datasource-options.factory.js';

export const createNestTypeOrmOptions = (
  envService: EnvService,
): TypeOrmModuleOptions => ({
  ...createDataSourceOptions(envService),

  autoLoadEntities: true,

  retryAttempts: 5,

  retryDelay: 3000,
});

===== database.module.ts =====
import {
  Global,
  Module,
} from '@nestjs/common';

import {
  TypeOrmModule,
} from '@nestjs/typeorm';

import {
  AppConfigModule,
  EnvService,
} from '../config/index.js';

import {
  createNestTypeOrmOptions,
} from './nest-typeorm-options.factory.js';

@Global()
@Module({
  imports: [
    TypeOrmModule.forRootAsync({
      imports: [
        AppConfigModule,
      ],

      inject: [
        EnvService,
      ],

      useFactory:
        createNestTypeOrmOptions,
    }),
  ],

  exports: [
    TypeOrmModule,
  ],
})
export class DatabaseModule {}
