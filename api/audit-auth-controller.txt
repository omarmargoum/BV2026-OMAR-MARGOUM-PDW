import {
  Body,
  Controller,
  Headers,
  HttpCode,
  HttpStatus,
  Get,
  Ip,
  Post,
  Res,
  UseGuards,
} from '@nestjs/common';
import type { Response,
} from 'express';
import {
  ApiBearerAuth,
  ApiConflictResponse,
  ApiCreatedResponse,
  ApiOkResponse,
  ApiOperation,
  ApiTags,
  ApiUnauthorizedResponse,
  ApiUnprocessableEntityResponse,
} from '@nestjs/swagger';

import {
  AccountResponseDto,
} from '../../account/data/index.js';
import {
  AccountService,
} from '../../account/index.js';
import type {
  AuthContext,
} from '../data/model/auth-context.js';
import {
  LoginRequestDto,
  LoginResponseDto,
  LogoutRequestDto,
  LogoutResponseDto,
  RegisterRequestDto,
  RefreshRequestDto, 
  RefreshResponseDto,
} from '../data/index.js';
import {
  CurrentAuth,
  Public,
} from '../decorator/index.js';

import {
  AuthCookieService,
  AuthService,
} from '../service/index.js';

@ApiTags('Authentication')
@Controller('auth')
export class AuthController {
  constructor(
    private readonly authService:
    AuthService,

    private readonly accountService:
    AccountService,

    private readonly authCookieService:
    AuthCookieService,
  ) {}

  @ApiOperation({
    summary:
      'Créer un compte utilisateur',
    description:
      'Crée le compte, l’identifiant de connexion et le credential de mot de passe dans une transaction unique.',
  })
  @ApiCreatedResponse({
    description:
      'Le compte a été créé avec succès.',
    type: AccountResponseDto,
  })
  @ApiConflictResponse({
    description:
      'L’identifiant de connexion est indisponible.',
  })
  @ApiUnprocessableEntityResponse({
    description:
      'Les données envoyées ne respectent pas les règles de validation.',
  })
  @Public()
  @HttpCode(HttpStatus.CREATED)
  @Post('register')
  async register(
    @Body()
    input: RegisterRequestDto,
  ): Promise<AccountResponseDto> {
    return this.authService
      .register(input);
  }

  @ApiOperation({
    summary:
      'Vérifier les identifiants de connexion',
    description:
      'Vérifie l’adresse e-mail et le mot de passe. La création de session et de jetons sera ajoutée ensuite.',
  })
  @ApiOkResponse({
    description:
      'Les identifiants sont valides et une session a été créée.',
    type: LoginResponseDto,
  })
  @ApiUnauthorizedResponse({
    description:
      'Les identifiants sont invalides.',
  })
  @ApiUnprocessableEntityResponse({
    description:
      'Les données envoyées ne respectent pas les règles de validation.',
  })
  @Public()
  @HttpCode(HttpStatus.OK)
  @Post('login')
  async login(
    @Body()
    input: LoginRequestDto,
    @Ip()
    ipAddress: string,
    @Headers('user-agent')
    userAgent: string | undefined,
    @Res({ passthrough: true })
    response: Response,
  ): Promise<AccountResponseDto> {
    const authenticatedSession = await this.authService
      .login(
      input,
      {
        ipAddress,
        userAgent:
          userAgent ??
          null,
      },
    );

    this.authCookieService
      .setAccessCookie(
        response,
        authenticatedSession.accessToken,
      );

    this.authCookieService
      .setRefreshCookie(
        response,
        authenticatedSession.refreshToken,
        authenticatedSession.refreshExpiresAt,
      );

    this.authCookieService
      .setCsrfCookie(
        response,
        authenticatedSession.csrfToken,
        authenticatedSession.refreshExpiresAt,
      );

    return this.accountService
  .findById(
    authenticatedSession
      .accountId,
  );
  }

  @ApiOperation({
    summary:
      'Renouveler les jetons d’authentification',
    description:
      'Invalide le refresh token actuel et délivre une nouvelle paire de jetons.',
  })
  @ApiOkResponse({
    description:
      'Les jetons ont été renouvelés.',
    type: RefreshResponseDto,
  })
  @ApiUnauthorizedResponse({
    description:
      'Le refresh token est invalide, expiré, utilisé ou révoqué.',
  })
  @ApiUnprocessableEntityResponse({
    description:
      'Le corps de la requête est invalide.',
  })
  @Public()
  @HttpCode(HttpStatus.OK)
  @Post('refresh')
  async refresh(
    @Body()
    input: RefreshRequestDto,
  ): Promise<RefreshResponseDto> {
    return this.authService
      .refresh(input);
  }
 @ApiOperation({
  summary:
    'Fermer une session utilisateur',
  description:
    'Révoque la session et tous les refresh tokens qui lui sont associés.',
})
@ApiOkResponse({
  description:
    'La déconnexion a été traitée.',
  type: LogoutResponseDto,
})
@ApiUnprocessableEntityResponse({
  description:
    'Le corps de la requête est invalide.',
})
@Public()
@HttpCode(HttpStatus.OK)
@Post('logout')
async logout(
  @Body()
  input: LogoutRequestDto,
): Promise<LogoutResponseDto> {
  return this.authService.logout(
    input,
  );
}
@ApiBearerAuth()
@ApiOperation({
  summary:
    'Vérifier la session authentifiée',
})
@ApiOkResponse({
  description:
    'Le jeton d’accès et la session sont valides.',
})
@ApiUnauthorizedResponse({
  description:
    'Le jeton ou la session est invalide.',
})

@Get('session')
getCurrentSession(
  @CurrentAuth()
  auth: AuthContext,
): AuthContext {
  return auth;
}
}
